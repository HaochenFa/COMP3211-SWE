% ---------------------------------------------------------------- requirements
% One requirement: the id hangs in the left margin column, the statement,
% rationale, and verification sit in the body column.
\newlength{\pimreqindent}
\setlength{\pimreqindent}{17mm}
\newenvironment{requirement}[2]{%
  \begin{list}{}{%
    \setlength{\leftmargin}{\pimreqindent}%
    \setlength{\labelwidth}{\pimreqindent}%
    \setlength{\labelsep}{0pt}%
    \setlength{\itemindent}{0pt}%
    \setlength{\topsep}{7pt}%
    \setlength{\parsep}{4pt}%
    \setlength{\listparindent}{0pt}%
  }%
  \item[{\makebox[\pimreqindent][l]{\sffamily\bfseries\color{accent}#1}}]%
  \ifstrempty{#2}{}{{\sffamily\bfseries\color{accent}#2}\par\nopagebreak}%
}{\end{list}}
\newenvironment{reqgroup}{%
  \begin{list}{}{%
    \setlength{\leftmargin}{\pimreqindent}%
    \setlength{\topsep}{4pt}%
    \setlength{\parsep}{3pt}%
  }\item\relax
}{\end{list}}
